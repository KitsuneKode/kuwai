cmake_minimum_required(VERSION 3.27)
project(kitty_sidebar_spike LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
find_package(PkgConfig REQUIRED)
pkg_check_modules(DEPS REQUIRED IMPORTED_TARGET hyprland libdrm libinput libudev pixman-1 wayland-server xkbcommon pangocairo)
add_library(kitty-sidebar SHARED sidebar.cpp)
target_link_libraries(kitty-sidebar PRIVATE PkgConfig::DEPS)
target_compile_options(kitty-sidebar PRIVATE -Wall -Wextra -Wpedantic)
add_executable(feed-test feed-test.cpp)
enable_testing()
add_test(NAME feed-contract COMMAND feed-test)
